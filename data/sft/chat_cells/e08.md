#   temp 0.0 -> modelin EN OLASI cevabi. Formulaik/nisan bir model hep
#               ayni klişe cevabi verir ve saklanir.
#   temp 0.7 -> cesitlilik. Hep ayni cevap geliyorsa ezberlemistir.
# Bir de EOS'a ulasip ulasmadigini sayiyoruz: ogrenmediyse cevap tasar
# ve devam eder.
import sys, torch
sys.path.insert(0, '/content/cofeuai/python')
from model import CofeuTransformer, ModelConfig
from tokenizer import BPETokenizer
import sft

CKPT = '/content/drive/MyDrive/cofeuai/sft_chat/cofeu_sft.pt'
ck = torch.load(CKPT, map_location='cpu', weights_only=False)
print("checkpoint iteration:", ck.get('iteration'), "| val:", ck.get('val_loss'))

DEV = "cuda" if torch.cuda.is_available() else "cpu"
print("cihaz:", DEV)

cfg = ModelConfig(**ck['config'].__dict__)
m = CofeuTransformer(cfg)
m.load_state_dict(ck['model_state'])
m.eval().to(DEV)
tok = BPETokenizer.load('/content/cofeuai/checkpoints/vocab.json')

# Sorgular kasten secildi:
#  - SELAMLAMA: sohbet modelinin ilk testi; genel klişe cevap riski
#  - BILGI: olgusal, dogru/yanlis net; genel kacinma yakalanir
#  - UZUNLUK: "üç kelimeyle" format talimati; sayma/uzunluk uyumu
#  - BILINMEYEN: uydurma riski; "bilmiyorum" demeyi ogrendi mi
#  - COZUM: hesap gerektirir; kalip degil gercek dusunme
SORULAR = [
    ("selamlama",  "Merhaba, nasılsın?"),
    ("bilgi",      "Türkiye'nin başkenti neresidir?"),
    ("uzunluk",    "Renkleri yalnızca üç kelimeyle yaz."),
    ("bilinmeyen", "Kaptan John Smith'in doğum tarihi nedir?"),
    ("cozum",      "Bir elma 5 TL, portakal 7 TL. 3 elma 2 portakal kaç TL?"),
]

def uret(prompt, temp):
    p, _ = sft.build_text(prompt, "", "")
    x = torch.tensor([tok.encode(p)], dtype=torch.long).to(DEV)
    torch.manual_seed(42)
    out = m.generate(x, 120, temperature=temp, top_k=40,
                     eos_token_id=tok.eos_id)
    yeni = out[0, len(x[0]):].tolist()
    eos_gordu = tok.eos_id in yeni
    metin = tok.decode([t for t in yeni if t != tok.eos_id])
    return metin, len(yeni), eos_gordu

for etiket, q in SORULAR:
    print(f"\n{'='*70}\n[{etiket}] {q}")
    for temp, ad in ((0.0, 'greedy'), (0.7, 'ornek')):
        metin, n, eos = uret(q, temp)
        print(f"  {ad:6s} ({n:3d} tok, EOS={'evet' if eos else 'YOK'}) {metin[:150]!r}")
